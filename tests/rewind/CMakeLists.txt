cmake_minimum_required(VERSION 3.20)
project(GeneratedRewindChecks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_library(check_lz4 STATIC ${ROOT}/vendor/lz4/lz4.c)
add_executable(rewind_checks checks.cpp ${ROOT}/src/rewind.cpp)
target_include_directories(rewind_checks PRIVATE ${ROOT}/src ${ROOT}/vendor/lz4)
target_link_libraries(rewind_checks PRIVATE check_lz4)
enable_testing()
add_test(NAME generated_rewind_bytes_audio_limits COMMAND rewind_checks)
