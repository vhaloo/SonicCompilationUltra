cmake_minimum_required(VERSION 3.20)
project(ProceduralVoxelChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(voxel_checks checks.cpp ../../src/voxel.cpp ../../src/voxel_export.cpp)
target_include_directories(voxel_checks PRIVATE ../../src)
add_executable(voxel_preview preview.cpp ../../src/voxel.cpp)
target_include_directories(voxel_preview PRIVATE ../../src)
if(MSVC)
  target_compile_options(voxel_checks PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME generated_voxel_geometry COMMAND voxel_checks)
set_tests_properties(generated_voxel_geometry PROPERTIES TIMEOUT 180)
