cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(SonicCompilationUltra VERSION 0.2.1 LANGUAGES C CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(FetchContent)
option(SCU_FETCH_SDL "Build pinned SDL2 when a system SDL2 is unavailable" ON)
set(SCU_SDL_ROOT "" CACHE PATH "Optional SDL2 development SDK directory")
if(SCU_SDL_ROOT AND WIN32)
  add_library(SDL2::SDL2 SHARED IMPORTED)
  set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_IMPLIB "${SCU_SDL_ROOT}/lib/x64/SDL2.lib" IMPORTED_LOCATION "${SCU_SDL_ROOT}/lib/x64/SDL2.dll" INTERFACE_INCLUDE_DIRECTORIES "${SCU_SDL_ROOT}/include")
else()
  find_package(SDL2 2.28 CONFIG QUIET)
  if(NOT TARGET SDL2::SDL2 AND SCU_FETCH_SDL)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sdl2/CMakeLists.txt")
      add_subdirectory(vendor/sdl2)
    else()
      set(SCU_SDL_ARCHIVE https://github.com/libsdl-org/SDL/releases/download/release-2.32.10/SDL2-2.32.10.tar.gz)
      if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2-2.32.10.tar.gz")
        set(SCU_SDL_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2-2.32.10.tar.gz")
      endif()
      FetchContent_Declare(sdl2 URL ${SCU_SDL_ARCHIVE} URL_HASH SHA256=5f5993c530f084535c65a6879e9b26ad441169b3e25d789d83287040a9ca5165)
      FetchContent_MakeAvailable(sdl2)
    endif()
  endif()
endif()
if(NOT TARGET SDL2::SDL2)
  message(FATAL_ERROR "SDL2 2.28+ is required")
endif()
include(cmake/SDLLegacyKeys.cmake)
include(cmake/GenesisCore.cmake)
add_library(scu_imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp vendor/imgui/backends/imgui_impl_sdl2.cpp vendor/imgui/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(scu_imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(scu_imgui PUBLIC SDL2::SDL2)
add_library(scu_disasm STATIC vendor/musashi-disassembler/m68kdasm.c)
target_compile_definitions(scu_disasm PRIVATE m68k_disassemble=scu_disassemble m68k_disassemble_raw=scu_disassemble_raw m68k_is_valid_instruction=scu_decode_valid)
add_library(scu_common STATIC src/rom.cpp src/world.cpp src/enemies.cpp src/spawn_catalogue.cpp src/rom_graphics.cpp src/rom_code.cpp src/settings.cpp src/localization.cpp)
target_include_directories(scu_common PUBLIC src vendor/genesis-plus-gx/libretro vendor/genesis-plus-gx/libretro/libretro-common/include vendor/zlib)
target_link_libraries(scu_common PUBLIC scu_core scu_disasm SDL2::SDL2)
find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
  target_link_libraries(scu_common PRIVATE OpenMP::OpenMP_CXX)
endif()
add_executable(SonicCompilationUltra src/main.cpp src/core.cpp src/ui.cpp)
target_link_libraries(SonicCompilationUltra PRIVATE scu_common scu_imgui)
target_compile_definitions(SonicCompilationUltra PRIVATE SDL_MAIN_HANDLED SCU_VERSION="${PROJECT_VERSION}")
if(MSVC)
  foreach(t scu_common SonicCompilationUltra scu_imgui)
    target_compile_options(${t} PRIVATE /utf-8 /W3 /permissive-)
  endforeach()
  target_compile_definitions(scu_core PRIVATE _CRT_SECURE_NO_WARNINGS __inline__=__inline)
  target_compile_options(scu_core PRIVATE /utf-8 /wd4244 /wd4267 /wd4018 /wd4996)
endif()
if(WIN32)
  configure_file(assets/SCU.rc.in ${CMAKE_CURRENT_BINARY_DIR}/SCU.rc @ONLY)
  target_sources(SonicCompilationUltra PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/SCU.rc)
  if(MSVC)
    target_link_options(SonicCompilationUltra PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
  else()
    target_link_options(SonicCompilationUltra PRIVATE -mwindows)
  endif()
  target_link_libraries(SonicCompilationUltra PRIVATE shell32)
  add_custom_command(TARGET SonicCompilationUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:SonicCompilationUltra>)
endif()
add_custom_command(TARGET SonicCompilationUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:SonicCompilationUltra>/assets)
install(TARGETS SonicCompilationUltra RUNTIME DESTINATION .)
if(WIN32)
  install(FILES $<TARGET_FILE:SDL2::SDL2> DESTINATION .)
  set(CMAKE_INSTALL_OPENMP_LIBRARIES TRUE)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
endif()
install(DIRECTORY assets DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION . OPTIONAL)
if(BUILD_TESTING)
  add_executable(scu_tests tests/tests.cpp tests/enemies.cpp)
  target_link_libraries(scu_tests PRIVATE scu_common)
  add_test(NAME scu_unit COMMAND scu_tests)
  add_executable(scu_runtime_tests tests/runtime.cpp src/core.cpp)
  target_link_libraries(scu_runtime_tests PRIVATE scu_common)
  target_compile_definitions(scu_runtime_tests PRIVATE SDL_MAIN_HANDLED)
  if(MSVC)
    target_compile_options(scu_runtime_tests PRIVATE /utf-8)
  endif()
  if(WIN32)
    add_custom_command(TARGET scu_runtime_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:scu_runtime_tests>)
  endif()
  add_test(NAME scu_runtime COMMAND scu_runtime_tests)
  add_executable(scu_ui_tests tests/ui_gamepad.cpp src/core.cpp src/ui.cpp)
  target_link_libraries(scu_ui_tests PRIVATE scu_common scu_imgui)
  target_compile_definitions(scu_ui_tests PRIVATE SDL_MAIN_HANDLED SCU_UI_TESTS SCU_VERSION="${PROJECT_VERSION}")
  if(MSVC)
    target_compile_options(scu_ui_tests PRIVATE /utf-8)
  endif()
  if(WIN32)
    add_custom_command(TARGET scu_ui_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:scu_ui_tests>)
  endif()
  add_test(NAME scu_gamepad_ui COMMAND scu_ui_tests)
  set_tests_properties(scu_gamepad_ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 300)
endif()
